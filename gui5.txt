local function create(className, properties, parent)
    local object = Instance.new(className)
    for key, value in pairs(properties) do
        object[key] = value
    end
    object.Parent = parent
    return object
end

local function round(parent, radius)
    create("UICorner", {CornerRadius = UDim.new(0, radius or 8)}, parent)
end

local function outline(parent, color, thickness, transparency)
    return create("UIStroke", {
        Color = color or Theme.border,
        Thickness = thickness or 1,
        Transparency = transparency or 0.12,
    }, parent)
end

local function pad(parent, left, right, top, bottom)
    create("UIPadding", {
        PaddingLeft = UDim.new(0, left or 0),
        PaddingRight = UDim.new(0, right or 0),
        PaddingTop = UDim.new(0, top or 0),
        PaddingBottom = UDim.new(0, bottom or 0),
    }, parent)
end

local function tween(object, duration, properties)
    local info = TweenInfo.new(duration or 0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
    local animation = TweenService:Create(object, info, properties)
    animation:Play()
    return animation
end

local function addParticle(parent, position, color)
    if not particlesEnabled then return end
    local dot = create("Frame", {
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = position,
        Size = UDim2.fromOffset(math.random(4, 7), math.random(4, 7)),
        BackgroundColor3 = color or Theme.primary,
        BorderSizePixel = 0,
        ZIndex = 100,
    }, parent)
    round(dot, 100)
    local dx = math.random(-30, 30)
    local dy = math.random(-34, -8)
    tween(dot, 0.38, {
        Position = position + UDim2.fromOffset(dx, dy),
        Size = UDim2.fromOffset(0, 0),
        BackgroundTransparency = 1,
    })
    task.delay(0.42, function()
        if dot and dot.Parent then dot:Destroy() end
    end)
end

local function burst(parent, absolutePosition, color)
    if not particlesEnabled then return end
    local localX = absolutePosition.X - parent.AbsolutePosition.X
    local localY = absolutePosition.Y - parent.AbsolutePosition.Y
    for _ = 1, 7 do
        addParticle(parent, UDim2.fromOffset(localX, localY), color)
    end
end

local function setPage(pageName)
    currentPage = pageName
    for name, page in pairs(pages) do
        page.Visible = name == pageName
    end
    for name, button in pairs(navButtons) do
        local active = name == pageName
        tween(button, 0.12, {
            BackgroundColor3 = active and Theme.primary or Theme.card,
            TextColor3 = active and Color3.new(1,1,1) or Theme.muted,
        })
        if active then
            local indicator = button:FindFirstChild("ActiveIndicator")
            if indicator then indicator.BackgroundColor3 = Theme.primary end
        end
    end
end

local function toast(message)
    local box = create("Frame", {
        AnchorPoint = Vector2.new(1, 0),
        Position = UDim2.new(1, -10, 0, 57),
        Size = UDim2.new(0, 210, 0, 34),
        BackgroundColor3 = Theme.card,
        BackgroundTransparency = 0.03,
        BorderSizePixel = 0,
        ZIndex = 90,
    }, toastLayer)
    round(box, 8)
    outline(box, Theme.primary, 1, 0.15)
    local label = create("TextLabel", {
        Position = UDim2.fromOffset(10, 0),
        Size = UDim2.new(1, -20, 1, 0),
        BackgroundTransparency = 1,
        Text = message,
        TextColor3 = Theme.text,
        Font = Enum.Font.Gotham,
        TextSize = 10,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextTruncate = Enum.TextTruncate.AtEnd,
        ZIndex = 91,
    }, box)
    box.BackgroundTransparency = 1
    label.TextTransparency = 1
    tween(box, 0.15, {BackgroundTransparency = 0.03})
    tween(label, 0.15, {TextTransparency = 0})
    task.delay(1.8, function()
        if box.Parent then
            tween(box, 0.18, {BackgroundTransparency = 1})
            tween(label, 0.18, {TextTransparency = 1})
            task.delay(0.2, function()
                if box.Parent then box:Destroy() end
            end)
        end
    end)
end

local function addToggle(parent, labelText, description, defaultValue, key)
    local height = description and 55 or 42
    local row = card(parent, height)
    create("TextLabel", {
        Position = UDim2.fromOffset(10, description and 7 or 0),
        Size = UDim2.new(1, -64, 0, description and 18 or height),
        BackgroundTransparency = 1,
        Text = labelText,
        TextColor3 = Theme.text,
        Font = Enum.Font.GothamMedium,
        TextSize = 11,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextYAlignment = description and Enum.TextYAlignment.Center or Enum.TextYAlignment.Center,
    }, row)
    if description then
        create("TextLabel", {
            Position = UDim2.fromOffset(10, 27),
            Size = UDim2.new(1, -20, 0, 17),
            BackgroundTransparency = 1,
            Text = description,
            TextColor3 = Theme.muted,
            Font = Enum.Font.Gotham,
            TextSize = 9,
            TextXAlignment = Enum.TextXAlignment.Left,
            TextTruncate = Enum.TextTruncate.AtEnd,
        }, row)
    end
    local toggle = create("TextButton", {
        AnchorPoint = Vector2.new(1, 0.5),
        Position = UDim2.new(1, -9, 0.5, 0),
        Size = UDim2.fromOffset(35, 19),
        BackgroundColor3 = defaultValue and Theme.primary or Theme.off,
        Text = "",
        AutoButtonColor = false,
        BorderSizePixel = 0,
    }, row)
    round(toggle, 20)
    local knob = create("Frame", {
        AnchorPoint = Vector2.new(0, 0.5),
        Position = defaultValue and UDim2.new(1, -17, 0.5, 0) or UDim2.new(0, 3, 0.5, 0),
        Size = UDim2.fromOffset(13, 13),
        BackgroundColor3 = Color3.new(1, 1, 1),
        BorderSizePixel = 0,
    }, toggle)
    round(knob, 20)
    controlValues[key or labelText] = defaultValue == true

    local function update(value, notify)
        controlValues[key or labelText] = value
        tween(toggle, 0.12, {BackgroundColor3 = value and Theme.primary or Theme.off})
        tween(knob, 0.12, {Position = value and UDim2.new(1, -16, 0.5, 0) or UDim2.new(0, 3, 0.5, 0)})
        if notify then
            burst(main, toggle.AbsolutePosition + toggle.AbsoluteSize / 2, Theme.primary)
            toast(labelText .. (value and " enabled (UI only)" or " disabled (UI only)"))
        end
    end
    toggle.Activated:Connect(function()
        update(not controlValues[key or labelText], true)
    end)
    return {row = row, set = update, get = function() return controlValues[key or labelText] end}
end

local function addSlider(parent, labelText, minValue, maxValue, defaultValue, key)
    local row = card(parent, 55)
    local value = defaultValue
    create("TextLabel", {
        Position = UDim2.fromOffset(10, 4),
        Size = UDim2.new(0.72, 0, 0, 17),
        BackgroundTransparency = 1,
        Text = labelText,
        TextColor3 = Theme.text,
        Font = Enum.Font.GothamMedium,
        TextSize = 10,
        TextXAlignment = Enum.TextXAlignment.Left,
    }, row)
    local valueLabel = create("TextLabel", {
        AnchorPoint = Vector2.new(1, 0),
        Position = UDim2.new(1, -10, 0, 4),
        Size = UDim2.new(0, 48, 0, 17),
        BackgroundTransparency = 1,
        Text = tostring(value),
        TextColor3 = Theme.primary,
        Font = Enum.Font.GothamBold,
        TextSize = 10,
        TextXAlignment = Enum.TextXAlignment.Right,
    }, row)
    local track = create("TextButton", {
        Position = UDim2.new(0, 10, 0, 32),
        Size = UDim2.new(1, -20, 0, 6),
        BackgroundColor3 = Theme.off,
        Text = "",
        AutoButtonColor = false,
        BorderSizePixel = 0,
    }, row)
    round(track, 5)
    local ratio = (value - minValue) / math.max(1, maxValue - minValue)
    local fill = create("Frame", {
        Size = UDim2.new(ratio, 0, 1, 0),
        BackgroundColor3 = Theme.primary,
        BorderSizePixel = 0,
    }, track)
    round(fill, 5)
    local knob = create("Frame", {
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(ratio, 0, 0.5, 0),
        Size = UDim2.fromOffset(12, 12),
        BackgroundColor3 = Theme.text,
        BorderSizePixel = 0,
    }, track)
    round(knob, 10)
    controlValues[key or labelText] = value
    local dragging = false
    local function setFromX(x)
        local r = math.clamp((x - track.AbsolutePosition.X) / math.max(1, track.AbsoluteSize.X), 0, 1)
        value = math.floor(minValue + r * (maxValue - minValue) + 0.5)
        controlValues[key or labelText] = value
        fill.Size = UDim2.new(r, 0, 1, 0)
        knob.Position = UDim2.new(r, 0, 0.5, 0)
        valueLabel.Text = tostring(value)
    end
    track.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            setFromX(input.Position.X)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            setFromX(input.Position.X)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    return row
end

local function partButton(name, x, y, w, h, radius)
    local button = create("TextButton", {
        Name = name,
        Position = UDim2.fromOffset(x, y),
        Size = UDim2.fromOffset(w, h),
        BackgroundColor3 = Theme.secondary,
        BorderSizePixel = 0,
        Text = "",
        AutoButtonColor = false,
    }, mannequin)
    round(button, radius or 7)
    outline(button, Theme.border, 1, 0.2)
    selectedParts[name] = false
    button.Activated:Connect(function()
        selectedParts[name] = not selectedParts[name]
        local active = selectedParts[name]
        tween(button, 0.12, {
            BackgroundColor3 = active and Theme.primary or Theme.secondary,
        })
        local line = button:FindFirstChildOfClass("UIStroke")
        if line then
            line.Color = active and Theme.text or Theme.border
            line.Thickness = active and 2 or 1
        end
        burst(main, button.AbsolutePosition + button.AbsoluteSize / 2, Theme.primary)
        toast(name .. (active and " selected" or " unselected") .. " (visual demo)")
    end)
    return button
end

    local names = {}
    for name, active in pairs(selectedParts) do

-- Keep the particles toggle in sync with the effect system.
local function refreshSettingsFromValues()
    particlesEnabled = controlValues["Particles"] ~= false
    rgbEnabled = controlValues["RGB"] == true
end

-- Update behavior when the user clicks the relevant settings toggles.
for _, page in pairs(pages) do
    for _, row in ipairs(page:GetChildren()) do
        if row:IsA("Frame") then
            local label = row:FindFirstChildOfClass("TextLabel")
            local button = row:FindFirstChildOfClass("TextButton")
            if label and button then
                if label.Text == "UI particles" then
                    button.Activated:Connect(function()
                        task.defer(function()
                            refreshSettingsFromValues()
                            toast("Particles " .. (particlesEnabled and "on" or "off"))
                        end)
                    end)
                elseif label.Text == "RGB mode" then
                    button.Activated:Connect(function()
                        task.defer(function()
                            refreshSettingsFromValues()
                            toast("RGB " .. (rgbEnabled and "enabled" or "disabled"))
                        end)
                    end)
                end
            end
        end
    end
end

-- RGB accent animation; only recolors UI accents.
local rgbConnection
rgbConnection = RunService.RenderStepped:Connect(function(dt)
    if closed or not gui.Parent then
        if rgbConnection then rgbConnection:Disconnect() end
        return
    end
    if rgbEnabled then
        currentRgbHue = (currentRgbHue + dt * 0.12) % 1
        local color = Color3.fromHSV(currentRgbHue, 0.68, 1)
        Theme.primary = color
        mainStroke.Color = color
        brandMark.BackgroundColor3 = color
        orbStroke.Color = color
        for _, page in pairs(pages) do page.ScrollBarImageColor3 = color end
        for name, button in pairs(navButtons) do
            if name == currentPage then button.BackgroundColor3 = color end
        end
    end
end)

-- Drag support for the main window. Touch and mouse both work.
local function makeDraggable(handle, target, allowTouch)
    local dragging = false
    local dragInput
    local startInputPosition
    local startPosition

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or (allowTouch and input.UserInputType == Enum.UserInputType.Touch) then
            dragging = true
            startInputPosition = input.Position
            startPosition = target.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    handle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input == dragInput or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - startInputPosition
            target.Position = UDim2.new(
                startPosition.X.Scale, startPosition.X.Offset + delta.X,
                startPosition.Y.Scale, startPosition.Y.Offset + delta.Y
            )
        end
    end)
end
makeDraggable(header, main, true)
makeDraggable(orb, orb, true)

local resizing = false
local resizeStart
local sizeStart
resizeGrip.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        resizing = true
        resizeStart = input.Position
        sizeStart = main.AbsoluteSize
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if resizing and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - resizeStart
        local width = math.clamp(sizeStart.X + delta.X, 315, math.min(900, workspace.CurrentCamera.ViewportSize.X - 12))
        local height = math.clamp(sizeStart.Y + delta.Y, 340, math.min(720, workspace.CurrentCamera.ViewportSize.Y - 12))
        main.Size = UDim2.fromOffset(width, height)
    end
end)
UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        resizing = false
    end
end)

-- Basic responsive size for narrow screens; still user-resizable afterward.
local camera = workspace.CurrentCamera
local function adaptToViewport()
    if not camera or isMinimized then return end
    local viewport = camera.ViewportSize
    if viewport.X < 520 then
        main.Size = UDim2.fromOffset(math.max(315, viewport.X - 12), math.min(470, viewport.Y - 24))
        main.Position = UDim2.fromScale(0.5, 0.5)
    end
end
adaptToViewport()
if camera then
    camera:GetPropertyChangedSignal("ViewportSize"):Connect(adaptToViewport)
end
